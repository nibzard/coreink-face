cmake_minimum_required(VERSION 3.16)
project(coreink_host LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INK_SANITIZE "Enable address/undefined behavior sanitizers" ON)
if(INK_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(coreink STATIC ../components/coreink_core/history.cpp ../components/coreink_core/coreink.cpp ../components/coreink_core/render.cpp ../components/coreink_core/cache.cpp ../components/coreink_core/http_body.cpp vendor/cJSON.c)
target_include_directories(coreink PUBLIC ../components/coreink_core/include PRIVATE vendor)
target_compile_options(coreink PRIVATE -Wall -Wextra -Werror)
set_source_files_properties(vendor/cJSON.c PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations")
add_executable(coreink_tests test_core.cpp)
target_link_libraries(coreink_tests PRIVATE coreink)
add_test(NAME coreink_behavior COMMAND coreink_tests)
add_executable(ink-preview preview.cpp)
target_link_libraries(ink-preview PRIVATE coreink)
